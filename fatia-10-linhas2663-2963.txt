async function listarDoDB(storeName) {
  if (!db) await initDB();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([storeName], 'readonly');
    const store = transaction.objectStore(storeName);
    const request = store.getAll();
    request.onsuccess = () => resolve(request.result);
    request.onerror = () => reject(request.error);
  });
}

async function deletarDoDB(storeName, key) {
  if (!db) await initDB();
  return new Promise((resolve, reject) => {
    const transaction = db.transaction([storeName], 'readwrite');
    const store = transaction.objectStore(storeName);
    const request = store.delete(key);
    request.onsuccess = () => resolve();
    request.onerror = () => reject(request.error);
  });
}

let autoSaveTimer = null;
function agendarAutoSave() {
  if (autoSaveTimer) clearTimeout(autoSaveTimer);
  autoSaveTimer = setTimeout(async () => {
    try {
      await sincronizarComDB();
      console.log('✅ Auto-save realizado');
    } catch(e) {
      console.error('Erro no auto-save:', e);
    }
  }, 3000);
}

async function sincronizarComDB() {
  if (!db) await initDB();
  
  // Salvar ementas
  for (const ementa of ementas) {
    await salvarNoDB('ementas', ementa);
  }
  
  // Salvar templates
  for (const template of templates) {
    await salvarNoDB('templates', template);
  }
  
  // Salvar histórico
  for (const item of historico) {
    await salvarNoDB('historico', item);
  }
  
  // Salvar configurações
  await salvarNoDB('config', { key: 'filtros', value: filtrosAtivos });
  await salvarNoDB('config', { key: 'formatacao', value: formatacaoAtual });
  await salvarNoDB('config', { key: 'acoes_custom', value: acoesCustom });
}

async function carregarDoDB() {
  if (!db) await initDB();
  
  try {
    // Carregar ementas
    const ementasDB = await listarDoDB('ementas');
    if (ementasDB.length > 0) {
      ementas = ementasDB;
      saveLS('aj_ementas', ementas);
    }
    
    // Carregar templates
    const templatesDB = await listarDoDB('templates');
    if (templatesDB.length > 0) {
      templates = templatesDB;
      saveLS('aj_templates', templates);
    }
    
    // Carregar histórico
    const historicoDB = await listarDoDB('historico');
    if (historicoDB.length > 0) {
      historico = historicoDB;
      saveLS('aj_historico', historico);
    }
    
    // Carregar configurações
    const configFiltros = await lerDoDB('config', 'filtros');
    if (configFiltros) {
      filtrosAtivos = configFiltros.value;
      saveLS('aj_filtros', filtrosAtivos);
    }
    
    const configAcoes = await lerDoDB('config', 'acoes_custom');
    if (configAcoes) {
      acoesCustom = configAcoes.value;
      saveLS('aj_acoes_custom', acoesCustom);
    }
    
    console.log('✅ Dados carregados do IndexedDB');
  } catch(e) {
    console.error('Erro ao carregar do DB:', e);
  }
}

async function exportarBackupCompleto() {
  try {
    const backup = {
      versao: '2.0',
      timestamp: new Date().toISOString(),
      dados: {
        ementas: ementas,
        templates: templates,
        historico: historico,
        acoesCustom: acoesCustom,
        filtros: filtrosAtivos,
        formatacao: formatacaoAtual,
        chaves: {
          resumo: loadLS('aj_key_resumo', ''),
          geracao: loadLS('aj_key_geracao', ''),
          modelo: loadLS('aj_modelo', ''),
          promptGlobal: loadLS('aj_prompt_global', '')
        },
        config: {
          effort: effortLevel,
          verbosity: verbosity,
          templateAtivo: templateAtivo
        }
      }
    };
    
    const json = JSON.stringify(backup, null, 2);
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `backup-assistente-juridico-${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
    
    alert('✅ Backup exportado com sucesso!');
  } catch(e) {
    alert('❌ Erro ao exportar backup: ' + e.message);
  }
}
