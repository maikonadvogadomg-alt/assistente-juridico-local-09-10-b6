async function testarAPICustom() {
  const url = document.getElementById('api-custom-url').value.trim();
  const header = document.getElementById('api-custom-header').value.trim();
  
  if (!url) {
    alert('Configure a URL primeiro.');
    return;
  }
  
  try {
    const headers = { 'Content-Type': 'application/json' };
    if (header) headers['Authorization'] = header;
    
    const resp = await fetch(url, {
      method: 'GET',
      headers: headers
    });
    
    if (resp.ok) {
      alert('✅ API respondeu com sucesso!');
    } else {
      alert(`⚠️ API respondeu com status ${resp.status}`);
    }
  } catch(e) {
    alert('❌ Erro: ' + e.message);
  }
}

async function enviarParaAPI(dados) {
  const url = loadLS('aj_api_url', '');
  const header = loadLS('aj_api_header', '');
  const method = loadLS('aj_api_method', 'POST');
  
  if (!url) {
    alert('Configure a API personalizada primeiro na aba Backup.');
    return;
  }
  
  try {
    const headers = { 'Content-Type': 'application/json' };
    if (header) headers['Authorization'] = header;
    
    const resp = await fetch(url, {
      method: method,
      headers: headers,
      body: JSON.stringify(dados)
    });
    
    if (!resp.ok) throw new Error(`Status ${resp.status}`);
    
    alert('✅ Documento enviado via API!');
  } catch(e) {
    alert('❌ Erro ao enviar: ' + e.message);
  }
}

async function limparTodosDados() {
  if (!confirm('⚠️ ATENÇÃO: Isso apagará TODOS os dados permanentemente!\n\nDeseja continuar?')) {
    return;
  }
  
  if (prompt('Digite "CONFIRMAR" para apagar tudo:') !== 'CONFIRMAR') {
    return;
  }
  
  try {
    // Limpar localStorage
    localStorage.clear();
    
    // Limpar IndexedDB
    if (db) {
      const stores = ['documentos', 'ementas', 'templates', 'config', 'historico'];
      for (const store of stores) {
        const transaction = db.transaction([store], 'readwrite');
        const objectStore = transaction.objectStore(store);
        objectStore.clear();
      }
    }
    
    // Resetar variáveis
    ementas = [];
    templates = [];
    historico = [];
    acoesCustom = [];
    emsSelected.clear();
    
    alert('✅ Todos os dados foram apagados. Recarregue a página.');
    location.reload();
  } catch(e) {
    alert('❌ Erro: ' + e.message);
  }
}

function limparCache() {
  if ('caches' in window) {
    caches.keys().then(names => {
      names.forEach(name => caches.delete(name));
    });
  }
  alert('✅ Cache limpo!');
}
